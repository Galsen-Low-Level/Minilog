80 * @brief detect wich section group of binary contain the first value
81 * e.g 0000.0000.0000.0010.0000.0111
82 * |xxxxxxxxxxxxx|--------------|

84 * This function is used by bc_binv2
85 * @param int - the value
86 * @return int - the section(th) index that contain the value

88static __inline__ int detect_bit_section_starting_group(int value)
89{
90
91 size_t vsize = _SIZE(value) ;
92 int group = 0 , section = 1, section_start_bit =0 ;
93 while( 0 < vsize )
94 {
95 if(group == 4)

97 group&=~group ;
98 section=-~section;

103
104 group=-~group ;
105 vsize+=~(vsize^vsize);
106 }

108 return section +(~0);
