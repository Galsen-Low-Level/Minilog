361static __always_inline int minilog_apply_lglvl(int __log_level)
362{
363 int what_happen = 0 ;
364 switch (__log_level)
365 {
366 case INFO : __get_lp_level(INFO) ;
367 what_happen = (MM_INFO <<4) ; break;
368 case WARN : __get_lp_level(WARN);
369 what_happen = (MM_WARNING<<4) ; break;
370 case ERROR : __get_lp_level(ERROR) ;
371 what_happen = (MM_ERROR <<4) ; break;
372 case NOTHING: __get_lp_level(NOTHING) ;
373 what_happen = (MM_NOSEV <<4) ; break;
374 /* ------------Special log level ---------------- */
375 case ALERT :
376 __get_lp_level(ALERT) ;
377 what_happen = (MM_ALERT <<4) | ALRT;
378 break;
379 case FATALITY :
380 __get_lp_level(FATALITY);
381 what_happen = (MM_HALT <<4) | FATL;
382 break;
383 default :
384 what_happen=~0 /*?*/;break;
385 }
386
387 return what_happen ;
388}

398static __always_inline int minilog_syncom(unsigned char __comtype)
399{
400
401 /*By default PIPE communication is used to sychronize terminal log & logfile*/
402 if(!__comtype)
403 goto _defcom;
404
405 int magic_check = (0x7 ^ (__comtype >> 4));
406 if(!magic_check || !magic_check ^ 4)
407 return (__comtype ^(0xf0)) ;
408
409_defcom:
410 return MINILOG_COM_PIPE ;
